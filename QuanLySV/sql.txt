CREATE TABLE STUDENT (
    STUDENTID VARCHAR(20) PRIMARY KEY,
    NAME VARCHAR(100) NOT NULL,
    SEX INT,               -- sex: 1 (male), 0 (female)
    BIRTHOFDATE DATE,
    BIRTHLOCAL VARCHAR(200),
    VNEID VARCHAR(12) UNIQUE,
    DATEOFISSUE DATE,
    LOCALOFISSUE VARCHAR(200),
    LOCAL VARCHAR(255),
    PLACEOFRESIDENCE VARCHAR(255),
    NUMBERPHONE VARCHAR(15),
    STATUS INT             -- 1 (active), 0 (inactive)
);
